cmake_minimum_required(VERSION 3.18)
project(Quetz VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
option(QUETZ_BUILD_RUNTIME "Build the SST element and QEMU plugin" ON)
set(SST_ROOT "/opt/sst" CACHE PATH "Installed SST core prefix")
set(QEMU_PLUGIN_INCLUDE_DIR "/opt/qemu/include" CACHE PATH "Directory containing qemu-plugin.h")
if(QUETZ_BUILD_RUNTIME)
  find_path(SST_INCLUDE_DIR sst/core/component.h HINTS "${SST_ROOT}/include" REQUIRED)
  find_path(SST_CONFIG_INCLUDE_DIR sst_config.h HINTS "${SST_ROOT}/include" "${SST_ROOT}/include/sst/core" REQUIRED)
  find_path(QEMU_PLUGIN_HEADER_DIR qemu-plugin.h HINTS "${QEMU_PLUGIN_INCLUDE_DIR}" REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  file(GLOB QUETZ_SOURCES CONFIGURE_DEPENDS src/*.cc)
  add_library(quetz MODULE ${QUETZ_SOURCES})
  target_include_directories(quetz PRIVATE src include "${SST_INCLUDE_DIR}" "${SST_CONFIG_INCLUDE_DIR}")
  target_compile_definitions(quetz PRIVATE HAVE_CONFIG_H QEMU_PLUGIN_INSTALL_DIR="${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBEXECDIR}")
  file(GLOB PLUGIN_SOURCES CONFIGURE_DEPENDS src/qemu_plugin/*.cpp)
  add_library(qemu_sst_plugin MODULE ${PLUGIN_SOURCES})
  target_include_directories(qemu_sst_plugin PRIVATE src include "${SST_INCLUDE_DIR}" "${SST_CONFIG_INCLUDE_DIR}" "${QEMU_PLUGIN_HEADER_DIR}")
  target_link_libraries(qemu_sst_plugin PRIVATE PkgConfig::GLIB)
  if(APPLE)
    target_link_options(quetz PRIVATE "-undefined" "dynamic_lookup")
    target_link_options(qemu_sst_plugin PRIVATE "-undefined" "dynamic_lookup")
  endif()
  install(TARGETS quetz LIBRARY DESTINATION lib/sst-elements-library)
  install(TARGETS qemu_sst_plugin LIBRARY DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
  configure_file(tools/quetz-register.in quetz-register @ONLY)
  install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/quetz-register" runtime/quetz-run DESTINATION bin)
  install(FILES LICENSE.md DESTINATION share/licenses/quetz)
  install(DIRECTORY include/ DESTINATION include)
  install(DIRECTORY src/ DESTINATION include/sst/elements/quetz FILES_MATCHING PATTERN "*.h")
  install(DIRECTORY tests/ DESTINATION share/quetz/tests PATTERN "_build" EXCLUDE PATTERN "__pycache__" EXCLUDE)
  install(DIRECTORY qemu-overlay include DESTINATION share/quetz
      PATTERN "__pycache__" EXCLUDE)
endif()
if(BUILD_TESTING)
  file(GLOB UNIT_SOURCES CONFIGURE_DEPENDS tests/unit/test_*.cc)
  foreach(source IN LISTS UNIT_SOURCES)
    get_filename_component(name "${source}" NAME_WE)
    if(name STREQUAL "test_config_manager" AND NOT EXISTS "${SST_ROOT}/include/sst/core/component.h")
      continue()
    endif()
    if(name STREQUAL "test_ipc_client" AND (NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT EXISTS "${SST_ROOT}/include/sst/core/component.h"))
      continue()
    endif()
    add_executable(${name} "${source}")
    target_include_directories(${name} PRIVATE src include tests/unit "${SST_ROOT}/include")
    if(name STREQUAL "test_config_manager")
      target_sources(${name} PRIVATE src/quetz_platform_profiles.cc)
    elseif(name STREQUAL "test_ipc_client")
      target_sources(${name} PRIVATE qemu-overlay/quetz_ipc_client.c)
    elseif(name STREQUAL "test_ipc_layout")
      target_sources(${name} PRIVATE tests/unit/ipc_layout_c.c)
    endif()
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cache_policy COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/unit/test_cache_policy.py")
    add_test(NAME runtime_runner COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime" -p "test_*.py")
  endif()
endif()
